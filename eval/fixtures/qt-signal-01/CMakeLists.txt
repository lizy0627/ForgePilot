cmake_minimum_required(VERSION 3.24)
project(ForgePilotFixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(CMAKE_AUTOMOC ON)
include(cmake/Subject.cmake)
add_executable(smoke tests/public_test.cpp)
target_link_libraries(smoke PRIVATE subject)
add_custom_command(TARGET smoke POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:Qt6::Core> $<TARGET_FILE_DIR:smoke>)
enable_testing()
add_test(NAME public_test COMMAND smoke)
